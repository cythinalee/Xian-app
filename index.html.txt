                const level = getLevel(points);
                const range = level.max - level.min;

                const progress = ((points - level.min) / range) * 100;
                return Math.min(100, Math.max(0, progress));

            function formatTime(isoStr) {
                try {
                    const d = new Date(isoStr);
                    const now = new Date();
                    const diffMs = now - d;
                    const diffMin = Math.floor(diffMs / 60000);
                    if (diffMin < 1) return '刚刚';
                    if (diffMin < 60) return diffMin + '分钟前';
                    const diffHr = Math.floor(diffMin / 60);
                    if (diffHr < 24) return diffHr + '小时前';
                    const diffDay = Math.floor(diffHr / 24);
                    if (diffDay < 7) return diffDay + '天前';
                    return d.toLocaleDateString('zh-CN', { month: 'short', day: 'numeric' });
                } catch (e) {
                    return '';
                }
            }

                if (!confirm('⚠️ 这将清除所有数据（积分、任务、奖励、记录），确定要继续吗？此操作不可恢复！')) return;
                if (!confirm('再次确认：真的要删除全部数据吗？')) return;
                appData = getDefaultData();
                saveData(appData);
                renderAll();
                $settingsModal.style.display = 'none';
                showToast('所有数据已重置', 'warning');
            });

            $clearHistoryBtn.addEventListener('click', function() {
                if (!confirm('确定要清空所有积分记录吗？（积分不会减少）')) return;
                appData.history = [];
                saveData(appData);
                renderHistory();
                showToast('记录已清空', 'warning');
            });

            document.querySelector('.points-display').addEventListener('click', function(e) {
                const rect = this.getBoundingClientRect();
                spawnSparks(rect.left + rect.width / 2, rect.top + rect.height / 2, 6);
                bumpPointsAnimation();
            });

            document.addEventListener('keydown', function(e) {
                if (e.key === 'Escape') {
                    $settingsModal.style.display = 'none';
                    closeRedeemModal();
                }
            });

            renderAll();

            if (isNewDay && appData.tasks.some(t => t.lastCompletedDate === null)) {
                setTimeout(() => showToast('新的一天开始啦！加油完成今日任务吧！🌞', 'success'), 500);
            }

            console.log('🌟 成长积分App已就绪！');
            console.log('📋 当前积分:', appData.totalPoints);
            console.log('📅 今日日期:', getToday());
        })();
